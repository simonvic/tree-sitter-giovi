================================================================================
empty list creation
================================================================================
=> []
--------------------------------------------------------------------------------
(program (returnStatement (list)))

================================================================================
list creation with one value
================================================================================
=> [a]
--------------------------------------------------------------------------------
(program (returnStatement (list
	(listEntry (identifier)))))

================================================================================
list creation with multiple values
================================================================================
=> [a b c]
--------------------------------------------------------------------------------
(program (returnStatement (list
	(listEntry (identifier))
	(listEntry (identifier))
	(listEntry (identifier)))))

================================================================================
list creation with multiple comma separated values
================================================================================
=> [a, b, c]
--------------------------------------------------------------------------------
(program (returnStatement (list
	(listEntry (identifier))
	(listEntry (identifier))
	(listEntry (identifier)))))

================================================================================
nested empty list creation
================================================================================
=> [[]]
--------------------------------------------------------------------------------
(program (returnStatement (list
	(listEntry (list)))))

================================================================================
nested list creation
================================================================================
=> [[a]]
--------------------------------------------------------------------------------
(program (returnStatement (list
	(listEntry (list
		(listEntry (identifier)))))))

================================================================================
nested multiple list creation
================================================================================
=> [[a], [b], [c]]
--------------------------------------------------------------------------------
(program (returnStatement (list
	(listEntry (list
		(listEntry (identifier))))
	(listEntry (list
		(listEntry (identifier))))
	(listEntry (list
		(listEntry (identifier)))))))
