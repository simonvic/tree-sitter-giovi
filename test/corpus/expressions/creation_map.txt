================================================================================
empty map
================================================================================
=> [=>]
--------------------------------------------------------------------------------
(program (statement_return (map)))

================================================================================
map with one entry
================================================================================
=> [a=>a]
--------------------------------------------------------------------------------
(program (statement_return (map
	(map_entry
		key: (identifier)
		val: (identifier))
)))

================================================================================
map with multiple entries
================================================================================
=> [a=>a b=>b c=>c]
--------------------------------------------------------------------------------
(program (statement_return (map
	(map_entry
		key: (identifier)
		val: (identifier))
	(map_entry
		key: (identifier)
		val: (identifier))
	(map_entry
		key: (identifier)
		val: (identifier))
)))

================================================================================
map with multiple comma separated entries
================================================================================
=> [a=>a, b=>b, c=>c]
--------------------------------------------------------------------------------
(program (statement_return (map
	(map_entry
		key: (identifier)
		val: (identifier))
	(map_entry
		key: (identifier)
		val: (identifier))
	(map_entry
		key: (identifier)
		val: (identifier))
)))

================================================================================
nested map as value
================================================================================
=> [a=>[b=>b]]
--------------------------------------------------------------------------------
(program (statement_return (map
	(map_entry
		key: (identifier)
		val: (map
			(map_entry
				key: (identifier)
				val: (identifier)))))))

================================================================================
nested map as key
================================================================================
=> [[a=>a]=>b]
--------------------------------------------------------------------------------
(program (statement_return (map
	(map_entry
		key: (map
			(map_entry
				key: (identifier)
				val: (identifier)))
		val: (identifier)))))

================================================================================
nested map as key and as value
================================================================================
=> [[a=>a]=>[b=>b]]
--------------------------------------------------------------------------------
(program (statement_return (map
	(map_entry
		key: (map
			(map_entry
				key: (identifier)
				val: (identifier)))
		val: (map
			(map_entry
				key: (identifier)
				val: (identifier)))))))

================================================================================
list of maps
================================================================================
=> [[a=>a] [b=>b] [c=>c]]
--------------------------------------------------------------------------------
(program (statement_return (list
	(list_entry (map
		(map_entry
			key: (identifier)
			val: (identifier))))
	(list_entry (map
		(map_entry
			key: (identifier)
			val: (identifier))))
	(list_entry (map
		(map_entry
			key: (identifier)
			val: (identifier)))))))
