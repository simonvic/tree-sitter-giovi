================================================================================
ref identifier
================================================================================
=> ::f
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef (identifier))))

================================================================================
ref with only body
================================================================================
=> ::{}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef (anonFunction (block)))))

================================================================================
ref with empty parameters
================================================================================
=> ::(){}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	referenced: (anonFunction
		signature: (functionSignature (formalParameters))
		body: (block)))))

================================================================================
ref with one parameter
================================================================================
=> ::(a Int){}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	referenced: (anonFunction
		signature: (functionSignature
			(formalParameters
				(formalParameter
					name: (identifier)
					type: (typeIdentifier))))
		body: (block)))))

================================================================================
ref with multiple parameters
================================================================================
=> ::(a Int, b Int){}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	referenced: (anonFunction
		signature: (functionSignature
			(formalParameters
				(formalParameter
					name: (identifier)
					type: (typeIdentifier))
				(formalParameter
					name: (identifier)
					type: (typeIdentifier))))
		body: (block)))))
