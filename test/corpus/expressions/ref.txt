================================================================================
ref identifier
================================================================================
=> ::f
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef (identifier))))

================================================================================
ref with only body
================================================================================
=> ::{}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef (block))))

================================================================================
ref with empty parameters
================================================================================
=> ::(){}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	(formalParameters)
	(block))))

================================================================================
ref with one parameter
================================================================================
=> ::(a Int){}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	(formalParameters
		(formalParameter
			(identifier)
			(typeIdentifier)))
	(block))))

================================================================================
ref with multiple parameters
================================================================================
=> ::(a Int, b Int){}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	(formalParameters
		(formalParameter
			(identifier)
			(typeIdentifier))
		(formalParameter
			(identifier)
			(typeIdentifier)))
	(block))))

================================================================================
ref with return type
================================================================================
=> ::() => Int {}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	(formalParameters)
	(typeIdentifier)
	(block))))

================================================================================
ref with one parameter and return type
================================================================================
=> ::(a Int) => Int {}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	(formalParameters
		(formalParameter
			(identifier)
			(typeIdentifier)))
	(typeIdentifier)
	(block))))

================================================================================
ref with multiple parameters
================================================================================
=> ::(a Int, b Int) => Int {}
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	(formalParameters
		(formalParameter
			(identifier)
			(typeIdentifier))
		(formalParameter
			(identifier)
			(typeIdentifier)))
	(typeIdentifier)
	(block))))
