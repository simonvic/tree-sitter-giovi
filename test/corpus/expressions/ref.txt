================================================================================
ref identifier
================================================================================
=> ::f
--------------------------------------------------------------------------------
(program (return_statement (expression_ref (identifier))))

================================================================================
ref with only body
================================================================================
=> ::{}
--------------------------------------------------------------------------------
(program (return_statement (expression_ref (anon_function (block)))))

================================================================================
ref with empty parameters
================================================================================
=> ::(){}
--------------------------------------------------------------------------------
(program (return_statement (expression_ref
	referenced: (anon_function
		signature: (function_signature (formal_parameters))
		body: (block)))))

================================================================================
ref with one parameter
================================================================================
=> ::(a Int){}
--------------------------------------------------------------------------------
(program (return_statement (expression_ref
	referenced: (anon_function
		signature: (function_signature
			(formal_parameters
				(formal_parameter
					name: (identifier)
					type: (type_qualified typename: (identifier)))))
		body: (block)))))

================================================================================
ref with multiple parameters
================================================================================
=> ::(a Int, b Int){}
--------------------------------------------------------------------------------
(program (return_statement (expression_ref
	referenced: (anon_function
		signature: (function_signature
			(formal_parameters
				(formal_parameter
					name: (identifier)
					type: (type_qualified typename: (identifier)))
				(formal_parameter
					name: (identifier)
					type: (type_qualified typename: (identifier)))))
		body: (block)))))
