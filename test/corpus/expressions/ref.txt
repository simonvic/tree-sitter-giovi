ref identifier
================================================================================
=> ::f
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef (identifier))))

================================================================================
ref qualified identifier
================================================================================
=> ::foo.foo.foo
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	(expressionDot
		(identifier)
		(identifier))
	(identifier))))

================================================================================
=> ::foo()@foo.foo
--------------------------------------------------------------------------------
(program (returnStatement (expressionRef
	(expressionAccess
		(invocation

			(actualParameters))
		(identifier))
	(identifier))))
