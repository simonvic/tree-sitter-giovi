================================================================================
ref identifier
================================================================================
=> ::f
--------------------------------------------------------------------------------
(program (statement (returnStatement (expression (expressionRef (identifier))))))

================================================================================
ref with only body
================================================================================
=> ::{}
--------------------------------------------------------------------------------
(program (statement (returnStatement (expression (expressionRef (block))))))

================================================================================
ref with empty parameters
================================================================================
=> ::(){}
--------------------------------------------------------------------------------
(program (statement (returnStatement (expression (expressionRef
	(formalParameters)
	(block))))))

================================================================================
ref with one parameter
================================================================================
=> ::(a Int){}
--------------------------------------------------------------------------------
(program (statement (returnStatement (expression (expressionRef
	(formalParameters
		(formalParameter
			(identifier)
			(type (typeIdentifier))))
	(block))))))

================================================================================
ref with multiple parameters
================================================================================
=> ::(a Int, b Int){}
--------------------------------------------------------------------------------
(program (statement (returnStatement (expression (expressionRef
	(formalParameters
		(formalParameter
			(identifier)
			(type (typeIdentifier)))
		(formalParameter
			(identifier)
			(type (typeIdentifier))))
	(block))))))

================================================================================
ref with return type
================================================================================
=> ::() => Int {}
--------------------------------------------------------------------------------
(program (statement (returnStatement (expression (expressionRef
	(formalParameters)
	(type (typeIdentifier))
	(block))))))

================================================================================
ref with one parameter and return type
================================================================================
=> ::(a Int) => Int {}
--------------------------------------------------------------------------------
(program (statement (returnStatement (expression (expressionRef
	(formalParameters
		(formalParameter
			(identifier)
			(type (typeIdentifier))))
	(type (typeIdentifier))
	(block))))))

================================================================================
ref with multiple parameters
================================================================================
=> ::(a Int, b Int) => Int {}
--------------------------------------------------------------------------------
(program (statement (returnStatement (expression (expressionRef
	(formalParameters
		(formalParameter
			(identifier)
			(type (typeIdentifier)))
		(formalParameter
			(identifier)
			(type (typeIdentifier))))
	(type (typeIdentifier))
	(block))))))
