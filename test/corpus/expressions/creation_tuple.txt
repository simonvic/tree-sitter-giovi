================================================================================
empty tuple
================================================================================
=> <>
--------------------------------------------------------------------------------
(program (statement_return
	(tuple)))

================================================================================
tuple with one field
================================================================================
=> <a>
--------------------------------------------------------------------------------
(program (statement_return
	(tuple (identifier))))

================================================================================
tuple with multiple fields
================================================================================
=> <a b c>
--------------------------------------------------------------------------------
(program (statement_return
	(tuple
		(identifier)
		(identifier)
		(identifier))))

================================================================================
tuple with multiple comma separated fields
================================================================================
=> <a, b, c>
--------------------------------------------------------------------------------
(program (statement_return
	(tuple
		(identifier)
		(identifier)
		(identifier))))

================================================================================
tuple with nested tuple
================================================================================
=> <<a b c>>
--------------------------------------------------------------------------------
(program (statement_return
	(tuple
		(tuple
			(identifier)
			(identifier)
			(identifier)))))

================================================================================
tuple with nested tuples
================================================================================
=> <<a b c> <d e f>>
--------------------------------------------------------------------------------
(program (statement_return
	(tuple
		(tuple
			(identifier)
			(identifier)
			(identifier))
		(tuple
			(identifier)
			(identifier)
			(identifier)))))

================================================================================
tuple with relational expression as field
================================================================================
=> <a b<c d>
--------------------------------------------------------------------------------
(program (statement_return
	(tuple
		(identifier)
		(expression_relational
			(identifier)
			(identifier))
		(identifier))))
