(program (statement_return (list

================================================================================

--------------------------------------------------------------------------------
(program (statement_return (list
	(list_entry (identifier))

================================================================================

--------------------------------------------------------------------------------

================================================================================

--------------------------------------------------------------------------------
(program (statement_return (list
	(list_entry (list

	(list_entry (list
		(list_entry (identifier))))
	(list_entry (list
		(list_entry (identifier))))
	(list_entry (list
		(list_entry (identifier)))))))
