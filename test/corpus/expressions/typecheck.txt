================================================================================
typecheck
================================================================================
=> a is Int
--------------------------------------------------------------------------------
(program
	(statement_return
		(expression_typecheck
			(identifier)
			(type_qualified (identifier)))))

================================================================================
negated typecheck
================================================================================
=> a is not Int
--------------------------------------------------------------------------------
(program
	(statement_return
		(expression_typecheck
			(identifier)
			(type_qualified (identifier)))))

================================================================================
typecheck with list
================================================================================
=> a is [Int]
--------------------------------------------------------------------------------
(program
	(statement_return
		(expression_typecheck
			(identifier)
			(type_list (type_qualified typename: (identifier))))))

================================================================================
typecheck with map
================================================================================
=> a is [Int:Int]
--------------------------------------------------------------------------------
(program
	(statement_return
		(expression_typecheck
			(identifier)
			(type_map
				key_type: (type_qualified typename: (identifier))
				value_type: (type_qualified typename: (identifier))))))
