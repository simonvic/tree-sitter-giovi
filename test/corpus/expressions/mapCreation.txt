empty map
================================================================================
=> [:]
--------------------------------------------------------------------------------
(program (returnStatement (map)))

================================================================================
map with one entry
================================================================================
=> [a:a]
--------------------------------------------------------------------------------
(program (returnStatement (map
	(mapEntry
		key: (identifier)
		val: (identifier))
)))

map with multiple entries
================================================================================
=> [a:a b:b c:c]
--------------------------------------------------------------------------------
(program (returnStatement (map
	(mapEntry
		key: (identifier)
		val: (identifier))
	(mapEntry
		key: (identifier)
		val: (identifier))
	(mapEntry
		key: (identifier)
		val: (identifier))
)))
