================================================================================
empty tuple
================================================================================
=> <>
--------------------------------------------------------------------------------
(program (return_statement
	(tuple)))


================================================================================
tuple with one field
================================================================================
=> <a>
--------------------------------------------------------------------------------
(program (return_statement
	(tuple (identifier))))


================================================================================
tuple with multiple fields
================================================================================
=> <a b c>
--------------------------------------------------------------------------------
(program (return_statement
	(tuple
		(identifier)
		(identifier)
		(identifier))))


================================================================================
tuple with multiple comma separated fields
================================================================================
=> <a, b, c>
--------------------------------------------------------------------------------
(program (return_statement
	(tuple
		(identifier)
		(identifier)
		(identifier))))


================================================================================
tuple with named field
================================================================================
=> <a:a>
--------------------------------------------------------------------------------
(program (return_statement
	(record
		name: (identifier)
		value: (identifier))))


================================================================================
tuple with multiple named fields
================================================================================
=> <a:a b:b c:c>
--------------------------------------------------------------------------------
(program (return_statement
	(record
		(identifier) (identifier)
		(identifier) (identifier)
		(identifier) (identifier))))

================================================================================
tuple with nested tuple
================================================================================
=> <<a b c>>
--------------------------------------------------------------------------------
(program (return_statement
	(tuple
		(tuple
			(identifier)
			(identifier)
			(identifier)))))
