================================================================================
empty tuple
================================================================================
=> <>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple)))


================================================================================
tuple with one field
================================================================================
=> <a>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields (identifier)))))


================================================================================
tuple with multiple fields
================================================================================
=> <a b c>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields
		(identifier)
		(identifier)
		(identifier)))))


================================================================================
tuple with multiple comma separated fields
================================================================================
=> <a, b, c>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields
		(identifier)
		(identifier)
		(identifier)))))


================================================================================
tuple with named field
================================================================================
=> <a:a>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields
		name: (identifier)
		value: (identifier)))))


================================================================================
tuple with multiple named fields
================================================================================
=> <a:a b:b c:c>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields
		(identifier) (identifier)
		(identifier) (identifier)
		(identifier) (identifier)))))

================================================================================
tuple with nested tuple
================================================================================
=> <<a b c>>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields
		(tuple (tupleFields
			(identifier)
			(identifier)
			(identifier)))))))

================================================================================
tuple with nested tuples
================================================================================
=> <<a b c> <d e f>>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields
		(tuple (tupleFields
			(identifier)
			(identifier)
			(identifier)))
		(tuple (tupleFields
			(identifier)
			(identifier)
			(identifier)))))))

================================================================================
tuple with relational expression as field
================================================================================
=> <a b<c d>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields
		(identifier)
		(expressionBinary
			(identifier)
			(identifier))
		(identifier)))))


================================================================================
tuple with relational expression as named field
================================================================================
=> <x:a y:b<c z:d>
--------------------------------------------------------------------------------
(program (returnStatement
	(tuple (tupleFields
		name: (identifier) value: (identifier)
		name: (identifier) value: (expressionBinary
			(identifier)
			(identifier))
		name: (identifier) value: (identifier)))))
