================================================================================
RTL to primitive destination
================================================================================
\out <- "hello world"
--------------------------------------------------------------------------------
(program (statement (redirection
	(primitiveDestination)
	(expression (literal (literalText))))))

================================================================================
RTL to var declaration
================================================================================
t Text <- "hello world"
--------------------------------------------------------------------------------
(program (statement (redirection
	(varDeclaration
		(identifier)
		(type (typeIdentifier)))
	(expression (literal (literalText))))))

================================================================================
RTL to var declaration with default
================================================================================
t Text default "foo" <- "hello world"
--------------------------------------------------------------------------------
(program (statement (redirection
	(varDeclaration
		(identifier)
		(type (typeIdentifier))
		(defaultDefinition (expression (literal (literalText)))))
	(expression (literal (literalText))))))
