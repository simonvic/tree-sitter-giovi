================================================================================
RTL to primitive destination
================================================================================
\out <- "hello world"
--------------------------------------------------------------------------------
(program (statement (redirection
	(primitiveDestination)
	(expression (literalText)))))

================================================================================
RTL to var declaration
================================================================================
t Text <- "hello world"
--------------------------------------------------------------------------------
(program (statement (redirection
	(varDeclaration
		(identifier)
		(type (typeIdentifier)))
	(expression (literalText)))))

================================================================================
RTL to var declaration with default
================================================================================
t Text default "foo" <- "hello world"
--------------------------------------------------------------------------------
(program (statement (redirection
	(varDeclaration
		(identifier)
		(type (typeIdentifier))
		(defaultDefinition (expression (literalText))))
	(expression (literalText)))))

================================================================================
LTR to primitive destination
================================================================================
"hello world" -> \out
--------------------------------------------------------------------------------
(program (statement (redirection
	(expression (literalText))
	(primitiveDestination))))

================================================================================
LTR to var declaration
================================================================================
"hello world" -> t Text
--------------------------------------------------------------------------------
(program (statement (redirection
	(expression (literalText))
	(varDeclaration
		(identifier)
		(type (typeIdentifier))))))

================================================================================
LTR to var declaration with default
================================================================================
"hello world" -> t Text default "foo"
--------------------------------------------------------------------------------
(program (statement (redirection
	(expression (literalText))
	(varDeclaration
		(identifier)
		(type (typeIdentifier))
		(defaultDefinition (expression (literalText)))))))
