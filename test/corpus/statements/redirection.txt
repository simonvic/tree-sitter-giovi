================================================================================
RTL to primitive destination
================================================================================
\out <- "hello world"
--------------------------------------------------------------------------------
(program (redirection
	(primitiveDestination)
	(literalText)))

================================================================================
RTL to var declaration
================================================================================
t Text <- "hello world"
--------------------------------------------------------------------------------
(program (redirection
	(varDeclaration
		(identifier)
		(type (typeIdentifier)))
	(literalText)))

================================================================================
RTL to var declaration with default
================================================================================
t Text default "foo" <- "hello world"
--------------------------------------------------------------------------------
(program (redirection
	(varDeclaration
		(identifier)
		(type (typeIdentifier))
		(defaultDefinition (literalText)))
	(literalText)))

================================================================================
LTR to primitive destination
================================================================================
"hello world" -> \out
--------------------------------------------------------------------------------
(program (redirection
	(literalText)
	(primitiveDestination)))

================================================================================
LTR to var declaration
================================================================================
"hello world" -> t Text
--------------------------------------------------------------------------------
(program (redirection
	(literalText)
	(varDeclaration
		(identifier)
		(type (typeIdentifier)))))

================================================================================
LTR to var declaration with default
================================================================================
"hello world" -> t Text default "foo"
--------------------------------------------------------------------------------
(program (redirection
	(literalText)
	(varDeclaration
		(identifier)
		(type (typeIdentifier))
		(defaultDefinition (literalText)))))
