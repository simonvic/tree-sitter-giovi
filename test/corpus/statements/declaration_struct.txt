================================================================================
empty struct declaration
================================================================================
struct Foo <>
--------------------------------------------------------------------------------
(program (declaration_struct
	typename: (identifier)
	body: (struct_body)))

================================================================================
empty struct declaration with parent
================================================================================
Bar + struct Foo <>
--------------------------------------------------------------------------------
(program (declaration_struct
	parent_type: (type_qualified
		typename: (identifier))
	typename: (identifier)
	body: (struct_body)))

================================================================================
empty struct declaration with qualified parent
================================================================================
bar.Bar + struct Foo <>
--------------------------------------------------------------------------------
(program (declaration_struct
	parent_type: (type_qualified
		(identifier)
		typename: (identifier))
	typename: (identifier)
	body: (struct_body)))

================================================================================
struct declaration with one field
================================================================================
struct Foo <i:Int>
--------------------------------------------------------------------------------
(program (declaration_struct
	typename: (identifier)
	body: (struct_body
		(declaration_field
			name: (identifier)
			type: (type_qualified
				typename: (identifier))))))

================================================================================
struct declaration with one field with default override
================================================================================
struct Foo <i:Int default 69>
--------------------------------------------------------------------------------
(program (declaration_struct
	typename: (identifier)
	body: (struct_body
		(declaration_field
			name: (identifier)
			type: (type_qualified
				typename: (identifier))
			default: (literal_integer)))))
