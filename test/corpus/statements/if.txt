================================================================================
if expression do block
================================================================================
if true do {}
--------------------------------------------------------------------------------
(program
	(ifStatement
		condition: (expression (literalBool))
		body: (block)))

================================================================================
if expression do statement
================================================================================
if true do 69 -> \out
--------------------------------------------------------------------------------
(program
	(ifStatement
		condition: (expression (literalBool))
		body: (redirection (expression (literalInteger)) (primitiveDestination))))

================================================================================
if expression do block else block
================================================================================
if true do {} else {}
--------------------------------------------------------------------------------
(program
	(ifStatement
		condition: (expression (literalBool))
		body: (block)
		elseBody: (block)))

================================================================================
if expression do statement else statement
================================================================================
if true do 69 -> \out else "hello" -> \out
--------------------------------------------------------------------------------
(program
	(ifStatement
		condition: (expression (literalBool))
		body: (redirection (expression (literalInteger)) (primitiveDestination))
		elseBody: (redirection (expression (literalText)) (primitiveDestination))))


================================================================================
nested if else as statement
================================================================================
if true do
	if true do {}
	else {}
else {}
--------------------------------------------------------------------------------
(program (ifStatement
	condition: (expression (literalBool))
	body: (ifStatement
		condition: (expression (literalBool))
		body: (block)
		elseBody: (block))
	elseBody: (block)))


================================================================================
if else chain
================================================================================
if true do {}
else if true do {}
else if true do {}
else {}

--------------------------------------------------------------------------------
(program (ifStatement
	condition: (expression (literalBool))
	body: (block)
	elseBody: (ifStatement
		condition: (expression (literalBool))
		body: (block)
		elseBody: (ifStatement
			condition: (expression (literalBool))
			body: (block)
			elseBody: (block)))))
