================================================================================
if expression do block
================================================================================
if true do {}
--------------------------------------------------------------------------------
(program
	(ifStatement
		condition: (literalBool)
		body: (block)))

================================================================================
if expression do statement
================================================================================
if true do 69 -> _out
--------------------------------------------------------------------------------
(program
	(ifStatement
		condition: (literalBool)
		body: (redirection (literalInteger) (reservedIdentifier))))

================================================================================
if expression do block else block
================================================================================
if true do {} else {}
--------------------------------------------------------------------------------
(program
	(ifStatement
		condition: (literalBool)
		body: (block)
		elseBody: (block)))

================================================================================
if expression do statement else statement
================================================================================
if true do 69 -> _out else "hello" -> _out
--------------------------------------------------------------------------------
(program
	(ifStatement
		condition: (literalBool)
		body: (redirection (literalInteger) (reservedIdentifier))
		elseBody: (redirection (literalText) (reservedIdentifier))))


================================================================================
nested if else as statement
================================================================================
if true do
	if true do {}
	else {}
else {}
--------------------------------------------------------------------------------
(program (ifStatement
	condition: (literalBool)
	body: (ifStatement
		condition: (literalBool)
		body: (block)
		elseBody: (block))
	elseBody: (block)))


================================================================================
if else chain
================================================================================
if true do {}
else if true do {}
else if true do {}
else {}

--------------------------------------------------------------------------------
(program (ifStatement
	condition: (literalBool)
	body: (block)
	elseBody: (ifStatement
		condition: (literalBool)
		body: (block)
		elseBody: (ifStatement
			condition: (literalBool)
			body: (block)
			elseBody: (block)))))
