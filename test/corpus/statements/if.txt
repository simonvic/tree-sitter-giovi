================================================================================
if expression do block
================================================================================
if true do {}
--------------------------------------------------------------------------------
(program
	(statement
		(ifStatement
			condition: (expression (literalBool))
			body: (block))))

================================================================================
if expression do statement
================================================================================
if true do 69 -> \out
--------------------------------------------------------------------------------
(program
	(statement
		(ifStatement
			condition: (expression (literalBool))
			body: (statement (redirection (expression (literalInteger)) (primitiveDestination))))))

================================================================================
if expression do block else block
================================================================================
if true do {} else {}
--------------------------------------------------------------------------------
(program
	(statement
		(ifStatement
			condition: (expression (literalBool))
			body: (block)
			elseBody: (block))))

================================================================================
if expression do statement else statement
================================================================================
if true do 69 -> \out else "hello" -> \out
--------------------------------------------------------------------------------
(program
	(statement
		(ifStatement
			condition: (expression (literalBool))
			body: (statement (redirection (expression (literalInteger)) (primitiveDestination)))
			elseBody: (statement (redirection (expression (literalText)) (primitiveDestination))))))


================================================================================
nested if else as statement
================================================================================
if true do
	if true do {}
	else {}
else {}
--------------------------------------------------------------------------------
(program (statement (ifStatement
	condition: (expression (literalBool))
	body: (statement (ifStatement
		condition: (expression (literalBool))
		body: (block)
		elseBody: (block)))
	elseBody: (block))))


================================================================================
if else chain
================================================================================
if true do {}
else if true do {}
else if true do {}
else {}

--------------------------------------------------------------------------------
(program (statement (ifStatement
	condition: (expression (literalBool))
	body: (block)
	elseBody: (statement (ifStatement
		condition: (expression (literalBool))
		body: (block)
		elseBody: (statement (ifStatement
			condition: (expression (literalBool))
			body: (block)
			elseBody: (block))))))))
