================================================================================
if expression do block
================================================================================
if true do {}
--------------------------------------------------------------------------------
(program
	(statement_if
		condition: (literal_bool)
		body: (block)))

================================================================================
if expression do statement
================================================================================
if true do 69 -> _out
--------------------------------------------------------------------------------
(program
	(statement_if
		condition: (literal_bool)
		body: (redirection (literal_integer) (reserved_identifier))))

================================================================================
if expression do block else block
================================================================================
if true do {} else {}
--------------------------------------------------------------------------------
(program
	(statement_if
		condition: (literal_bool)
		body: (block)
		else_body: (block)))

================================================================================
if expression do statement else statement
================================================================================
if true do 69 -> _out else "hello" -> _out
--------------------------------------------------------------------------------
(program
	(statement_if
		condition: (literal_bool)
		body: (redirection (literal_integer) (reserved_identifier))
		else_body: (redirection (literal_text) (reserved_identifier))))


================================================================================
nested if else as statement
================================================================================
if true do
	if true do {}
	else {}
else {}
--------------------------------------------------------------------------------
(program (statement_if
	condition: (literal_bool)
	body: (statement_if
		condition: (literal_bool)
		body: (block)
		else_body: (block))
	else_body: (block)))


================================================================================
if else chain
================================================================================
if true do {}
else if true do {}
else if true do {}
else {}

--------------------------------------------------------------------------------
(program (statement_if
	condition: (literal_bool)
	body: (block)
	else_body: (statement_if
		condition: (literal_bool)
		body: (block)
		else_body: (statement_if
			condition: (literal_bool)
			body: (block)
			else_body: (block)))))
