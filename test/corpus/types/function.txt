================================================================================
function variable simple
================================================================================
f ::()
--------------------------------------------------------------------------------
(program (declaration_variable
	(identifier)
	(type_function_ref (function_signature (formal_parameters)))))

================================================================================
function variable with one argument
================================================================================
f ::(a Int)
--------------------------------------------------------------------------------
(program (declaration_variable
	(identifier)
	(type_function_ref
		(function_signature
			(formal_parameters
				(formal_parameter
					(identifier)
					(type_qualified (identifier))))))))
