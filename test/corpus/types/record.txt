================================================================================
empty record type
================================================================================
foo record <>
--------------------------------------------------------------------------------
(program
	(declaration_variable
		(identifier)
		(type_record)))

================================================================================
record type with one field
================================================================================
foo record <x Int>
--------------------------------------------------------------------------------
(program
	(declaration_variable
		(identifier)
		(type_record
			(declaration_field (identifier) (type_qualified (identifier))))))


================================================================================
record type with multiple comma separated fields
================================================================================
foo record <x Int, y Int, z Int,>
--------------------------------------------------------------------------------
(program
	(declaration_variable
		(identifier)
		(type_record
			(declaration_field (identifier) (type_qualified (identifier)))
			(declaration_field (identifier) (type_qualified (identifier)))
			(declaration_field (identifier) (type_qualified (identifier))))))

			(declaration_field (identifier) (type_qualified (identifier))))))
